        function handleFileUpload(e) {

            const files = e.target.files;

            if (!files.length) return;


            const today = new Date().toISOString().split('T')[0];


            Array.from(files).forEach(file => {

                let type = 'DOC';

                if (file.name.endsWith('.pdf')) type = 'PDF';

                if (file.name.endsWith('.ppt') || file.name.endsWith('.pptx')) type = 'PPT';


                const newMedia = {

                    id: Date.now() + Math.random(),

                    name: file.name,

                    size: (file.size / (1024 * 1024)).toFixed(2) + ' MB',

                    type: type,

                    date: today

                };


                appState.media[appState.activePractice].push(newMedia);

            });


            renderMediaList();

            saveToLocalStorage();

            e.target.value = '';

        }


        function addUrlMedia() {

            const titleInput = document.getElementById('media-title-input');

            const urlInput = document.getElementById('media-url-input');


            const title = titleInput.value.trim();

            const url = urlInput.value.trim();


            if (!title || !url) {

                alert('Harap isi nama bahan dan URL tautan!');

                return;

            }


            const today = new Date().toISOString().split('T')[0];

            const newMedia = {

                id: Date.now(),

                name: title,

                size: 'URL Link',

                type: 'LINK',

                date: today

            };


            appState.media[appState.activePractice].push(newMedia);

            titleInput.value = '';

            urlInput.value = '';


            renderMediaList();

            saveToLocalStorage();

        }

        function getRatingDescriptor(val) {

            const desc = { 1: 'Sangat Kurang', 2: 'Kurang', 3: 'Cukup', 4: 'Baik', 5: 'Sangat Baik' };

            return desc[val] || '';

        }

            const currentEval = appState.evaluations[appState.activePractice][appState.activeTab];

        function resetCurrentTabForm() {

            if (!confirm('Apakah Anda yakin ingin mengosongkan nilai form ini?')) return;

            const currentEval = appState.evaluations[appState.activePractice][appState.activeTab];

            RUBRIC_CRITERIA.forEach(c => currentEval.scores[c.id] = 0);

            currentEval.notes = '';

            renderRubricForm();

            updateAllSummaries();

            saveToLocalStorage();

        }


        function calculateAverageScore(scoresObj) {

            const vals = Object.values(scoresObj || {});

            if (!vals.length) return 0;

            const sum = vals.reduce((a, b) => a + b, 0);

            return (sum / RUBRIC_CRITERIA.length);

        }


        function getPredicate(score) {

            if (score >= 4.5) return 'Sangat Baik (A)';

            if (score >= 3.5) return 'Baik (B)';

            if (score >= 2.5) return 'Cukup (C)';

            if (score >= 1.5) return 'Kurang (D)';

            if (score > 0) return 'Sangat Kurang (E)';

            return 'Belum Dinilai';

        }


        function updateLiveTabScore() {

            const currentEval = appState.evaluations[appState.activePractice][appState.activeTab];

            const avg = calculateAverageScore(currentEval.scores);


            document.getElementById('live-score-val').innerText = avg.toFixed(1);

            document.getElementById('live-predikat-text').innerText = getPredicate(avg);

        }


        function updateAllSummaries() {

            const currentPracticeData = appState.evaluations[appState.activePractice];


            const selfAvg = calculateAverageScore(currentPracticeData.self.scores);

            const peerAvg = calculateAverageScore(currentPracticeData.peer.scores);

            const dosenAvg = calculateAverageScore(currentPracticeData.dosen.scores);


            // Update Cards

            document.getElementById('rekap-score-self').innerText = selfAvg.toFixed(2);

            document.getElementById('rekap-predikat-self').innerText = getPredicate(selfAvg);

            document.getElementById('bar-self').style.width = `${(selfAvg / 5) * 100}%`;


            document.getElementById('rekap-score-peer').innerText = peerAvg.toFixed(2);

            document.getElementById('rekap-predikat-peer').innerText = getPredicate(peerAvg);

            document.getElementById('bar-peer').style.width = `${(peerAvg / 5) * 100}%`;


            document.getElementById('rekap-score-dosen').innerText = dosenAvg.toFixed(2);

            document.getElementById('rekap-predikat-dosen').innerText = getPredicate(dosenAvg);

            document.getElementById('bar-dosen').style.width = `${(dosenAvg / 5) * 100}%`;


            renderComparisonTable();

            updateRadarChart();

        }
